                <h1>Глава 12. События в JavaScript</h1>

                <h2>Что такое события?</h2>
                <p>События — это действия, которые происходят в браузере: клики мышью, нажатия клавиш, загрузка страницы и т.д. JavaScript позволяет реагировать на эти события и выполнять код в ответ на действия пользователя.</p>

                <div class="info-box note">
                    <div class="info-box-title">💡 Основные типы событий</div>
                    <ul>
                        <li><strong>События мыши:</strong> click, dblclick, mouseover, mouseout, mousemove</li>
                        <li><strong>События клавиатуры:</strong> keydown, keyup, keypress</li>
                        <li><strong>События форм:</strong> submit, change, focus, blur, input</li>
                        <li><strong>События документа:</strong> DOMContentLoaded, load, scroll, resize</li>
                    </ul>
                </div>

                <h2>Способы назначения обработчиков событий</h2>

                <h3>1. HTML-атрибуты (не рекомендуется)</h3>
                <div class="code-block"><code>&lt;button onclick="alert('Clicked!')"&gt;Нажми меня&lt;/button&gt;

&lt;button onclick="handleClick()"&gt;Кнопка&lt;/button&gt;
&lt;script&gt;
function handleClick() {
    alert('Обработчик вызван!');
}
&lt;/script&gt;</code></div>

                <div class="info-box warning">
                    <div class="info-box-title">⚠️ Почему не рекомендуется?</div>
                    <p>Смешивание HTML и JavaScript делает код трудным для поддержки. Современный подход — разделять структуру (HTML) и поведение (JavaScript).</p>
                </div>

                <h3>2. DOM-свойства</h3>
                <div class="code-block"><code>const button = document.querySelector('button');

// Назначить обработчик
button.onclick = function() {
    alert('Кнопка нажата!');
};

// Можно использовать именованную функцию
function handleClick() {
    alert('Клик!');
}

button.onclick = handleClick;

// Удалить обработчик
button.onclick = null;</code></div>

                <div class="info-box warning">
                    <div class="info-box-title">⚠️ Ограничение</div>
                    <p>К одному событию можно привязать только ОДНУ функцию. При повторном присваивании предыдущая функция перезаписывается.</p>
                </div>

                <h3>3. addEventListener (рекомендуется)</h3>
                <div class="code-block"><code>const button = document.querySelector('button');

// Добавить обработчик
button.addEventListener('click', function() {
    alert('Первый обработчик');
});

// Можно добавить несколько обработчиков
button.addEventListener('click', function() {
    alert('Второй обработчик');
});

// С именованной функцией
function handleClick() {
    console.log('Клик обработан');
}

button.addEventListener('click', handleClick);

// Удалить обработчик (нужна ссылка на функцию)
button.removeEventListener('click', handleClick);</code></div>

                <div class="info-box success">
                    <div class="info-box-title">✅ Преимущества addEventListener</div>
                    <ul>
                        <li>Можно назначить несколько обработчиков на одно событие</li>
                        <li>Можно удалять обработчики</li>
                        <li>Поддержка фазы захвата (capture)</li>
                        <li>Работает с любыми событиями</li>
                    </ul>
                </div>

    console.log('Кнопка нажата');

box.addEventListener('mouseup', (e) => {
    console.log('Кнопка отпущена');

// contextmenu - правый клик (контекстное меню)

                <h2>Всплытие и погружение событий</h2>
                <p>События в DOM проходят три фазы:</p>
                <ol>
                    <li><strong>Фаза захвата (capturing)</strong> — от window к целевому элементу</li>
                    <li><strong>Фаза цели (target)</strong> — на самом элементе</li>
                    <li><strong>Фаза всплытия (bubbling)</strong> — от целевого элемента к window</li>
                </ol>

                <div class="code-block"><code>&lt;div class="outer"&gt;
    &lt;div class="inner"&gt;
        &lt;button&gt;Кликни&lt;/button&gt;
    &lt;/div&gt;
&lt;/div&gt;

&lt;script&gt;
const outer = document.querySelector('.outer');
const inner = document.querySelector('.inner');
const button = document.querySelector('button');

// Всплытие (по умолчанию)
outer.addEventListener('click', () => {
    console.log('1. Outer (всплытие)');
});

inner.addEventListener('click', () => {
    console.log('2. Inner (всплытие)');
});

button.addEventListener('click', () => {
    console.log('3. Button (цель)');
});

// При клике на button:
// 3. Button (цель)
// 2. Inner (всплытие)
// 1. Outer (всплытие)

// Захват (capture: true)
outer.addEventListener('click', () => {
    console.log('Outer (захват)');
}, true);

// При клике на button:
// Outer (захват)
// 3. Button (цель)
// 2. Inner (всплытие)
// 1. Outer (всплытие)
&lt;/script&gt;</code></div>

                <h3>event.target vs event.currentTarget</h3>
                <div class="code-block"><code>outer.addEventListener('click', (e) => {

                <h2>Остановка всплытия</h2>
                <div class="code-block"><code>button.addEventListener('click', (e) => {
    e.stopPropagation(); // Остановить всплытие
    console.log('Button clicked');
    // Обработчики на inner и outer НЕ сработают
});

// stopImmediatePropagation - остановить все обработчики
button.addEventListener('click', (e) => {
    e.stopImmediatePropagation();
    console.log('Первый обработчик');
});

button.addEventListener('click', (e) => {
    console.log('Второй обработчик'); // НЕ выполнится
});</code></div>

                <div class="info-box warning">
                    <div class="info-box-title">⚠️ Когда НЕ нужно останавливать всплытие</div>
                    <p>Остановка всплытия может сломать аналитику, модальные окна и другие системы, которые слушают события на document. Используйте только когда точно понимаете, что делаете.</p>
                </div>

                <h2>Делегирование событий</h2>
                <p>Вместо назначения обработчика каждому элементу, назначаем один обработчик родителю и используем <code>event.target</code>.</p>

                <div class="code-block"><code>&lt;ul id="menu"&gt;
    &lt;li&gt;Пункт 1&lt;/li&gt;
    &lt;li&gt;Пункт 2&lt;/li&gt;
    &lt;li&gt;Пункт 3&lt;/li&gt;
&lt;/ul&gt;

&lt;script&gt;
// Плохо: обработчик на каждый элемент
const items = document.querySelectorAll('li');
items.forEach(item => {
    item.addEventListener('click', () => {
        console.log('Клик на:', item.textContent);
    });
});

// Хорошо: один обработчик на родителе
const menu = document.querySelector('#menu');
menu.addEventListener('click', (e) => {
    if (e.target.tagName === 'LI') {
        console.log('Клик на:', e.target.textContent);
    }
});
&lt;/script&gt;</code></div>

                <h3>Практический пример</h3>
                <div class="code-block"><code>// Создаём событие "покупка товара"
class ShoppingCart {
    constructor() {
        this.items = [];
    }
    
    addItem(item) {
        this.items.push(item);
        
        // Генерируем событие
        const event = new CustomEvent('itemAdded', {
            detail: { item, total: this.items.length }
        });
        
        document.dispatchEvent(event);
    }
}

// Слушаем событие
document.addEventListener('itemAdded', (e) => {
    console.log(`Добавлен: ${e.detail.item}`);
    console.log(`Всего товаров: ${e.detail.total}`);
});

// Использование
const cart = new ShoppingCart();
cart.addItem('Ноутбук');  // Событие сработает
cart.addItem('Мышь');     // Событие сработает</code></div>

                <h2>Оптимизация обработчиков</h2>

                <h3>Debounce (задержка)</h3>
                <div class="code-block"><code>// Функция будет вызвана только через 300мс после последнего события
function debounce(func, delay) {
    let timeout;
    return function(...args) {
        clearTimeout(timeout);
        timeout = setTimeout(() => func.apply(this, args), delay);
    };
}

// Использование
const input = document.querySelector('input');

input.addEventListener('input', debounce((e) => {
    console.log('Поиск:', e.target.value);
}, 300));</code></div>

                <h3>Throttle (ограничение частоты)</h3>
                <div class="code-block"><code>// Функция будет вызываться не чаще, чем раз в delay мс
function throttle(func, delay) {
    let lastCall = 0;
    return function(...args) {
        const now = Date.now();
        if (now - lastCall >= delay) {
            lastCall = now;
            func.apply(this, args);
        }
    };
}

// Использование
window.addEventListener('scroll', throttle(() => {
    console.log('Прокрутка');
}, 100));</code></div>

                <div class="info-box note">
                    <div class="info-box-title">💡 Когда использовать?</div>
                    <ul>
                        <li><strong>Debounce:</strong> поиск при вводе, изменение размера окна</li>
                        <li><strong>Throttle:</strong> прокрутка, движение мыши, анимации</li>
                    </ul>
